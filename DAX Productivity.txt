Worker Count = 
VAR __SelectedWeek = SELECTEDVALUE(dimWeek[WeekStartDate])
VAR __WeekDates =
    SELECTCOLUMNS(CALENDAR(__SelectedWeek, __SelectedWeek + 6), "Date", [Date])
VAR __WeekFilter = TREATAS(__WeekDates, dimDate[Date])

RETURN
IF(
    ISBLANK(__SelectedWeek),
    BLANK(),
    CALCULATE(
        DISTINCTCOUNT(factVisitTimeDetail[WorkerId]),
        __WeekFilter
    )
)


Weeks Worked = 
VAR SelectedWeek =
    SELECTEDVALUE(dimWeek[WeekStartDate])

RETURN
IF(
    ISBLANK(SelectedWeek),
    BLANK(),
    1
)


Total Miles = 
SUMX(
    factVisitTimeDetail,
    COALESCE(factVisitTimeDetail[TotalMileage], 0)
)


Service Points = 
VAR SelectedWeek = SELECTEDVALUE(dimWeek[WeekStartDate])
VAR WeekDates =
    SELECTCOLUMNS(
        CALENDAR(SelectedWeek, SelectedWeek + 6),
        "Date", [Date]
    )
VAR WeekFilter = 
    TREATAS(WeekDates, dimDate[Date])
RETURN
IF(
    ISBLANK(SelectedWeek),
    BLANK(),
    CALCULATE(
        SUMX(
            factVisitTimeDetail,
            RELATED(dimHCHBServiceCodes[Point Value])
        ),
        factVisitTimeDetail[VisitStatus] = "Completed",
        WeekFilter
    )
)

Non - Admits = 
CALCULATE(
    DISTINCTCOUNT( factVisitTimeDetail[VisitId] ),
    dimHCHBServiceCodes[ScCode] = "RN00H",
    factVisitTimeDetail[VisitStatus] = "NON-ADMIT",
    USERELATIONSHIP( factVisitTimeDetail[VisitDate], dimDate[Date] )
)


Mileage Points = 
VAR Miles = [Total Miles]
RETURN
IF( Miles < 300, 0, 1 + INT( (Miles - 300) / 75 ) )

Report Notice = VAR __DISTINCT_VALUES_COUNT =
     DISTINCTCOUNT
     (
          'PowerBINotices'[Notice Text]
     )
VAR __MAX_VALUES_TO_SHOW = 2
RETURN
     COALESCE
     (
          IF
          (
               __DISTINCT_VALUES_COUNT
                    > __MAX_VALUES_TO_SHOW,
               CONCATENATE
               (
                    CONCATENATEX
                    (
                         TOPN
                         (
                              __MAX_VALUES_TO_SHOW,
                              VALUES( 'PowerBINotices'[Notice Text] ),
                              'PowerBINotices'[Notice Text], ASC
                         ),
                         'PowerBINotices'[Notice Text],
                         " | ",
                         'PowerBINotices'[Notice Text], ASC
                    ),
                    ", etc."
               ),
               CONCATENATEX
               (
                    VALUES( 'PowerBINotices'[Notice Text] ),
                    'PowerBINotices'[Notice Text],
                    " | ",
                    'PowerBINotices'[Notice Text], ASC
               )
          ),
          ""
     )

Avg Visits / Worker = 
AVERAGEX (
    VALUES ( dimWeek[WeekStartDate] ),
    DIVIDE (
        CALCULATE ( [Billable Visits] ),
        CALCULATE ( [Worker Count] )))

Avg Visits per Day = 
AVERAGEX(VALUES(dimWeek[WeekStartDate]),
DIVIDE([Billable Visits], [Days Worked])
)

Admissions = 
CALCULATE(
    DISTINCTCOUNT( factVisitTimeDetail[ScheduledVisitId] ),
    dimHCHBServiceCodes[ScVisitType] = "Hospice SOC",
    USERELATIONSHIP( factVisitTimeDetail[VisitDate], dimDate[Date] )
)


Billable Visits = 
COUNTROWS (
    FILTER (
        VALUES ( factVisitTimeDetail[ScheduledVisitId] ),
        CALCULATE (
            MAX ( factVisitTimeDetail[VisitStatus] ) = "COMPLETED"
                && MAX ( factVisitTimeDetail[Billable] ) = 1)))


Goal = 
SUMX(
    VALUES( factVisitTimeDetail[WorkerId] ),
    VAR WorkerWeeks = [Weeks Worked]
    VAR WeeklyGoal =
        CALCULATE( MAX( dimWorker[Goals] ) )
    RETURN
        WeeklyGoal * WorkerWeeks
)
