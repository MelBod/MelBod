Episodes Ending = 
CALCULATE (
    DISTINCTCOUNT ( FactTaskProductivity_HH[Episode ID] ),
    FactTaskProductivity_HH[OpenClosedFlag] = "Closed",
    FactTaskProductivity_HH[CompletedVisits] > 0,
    NOT ( ISBLANK ( FactTaskProductivity_HH[CompletedVisits] ) ),
    USERELATIONSHIP (
        FactTaskProductivity_HH[skepisodeEndDateKey],
        DimDates[DateKey]
    )
)

Non Billable Visits = CALCULATE (
    SUM ( FactTaskProductivity_HH[CompletedVisits] ),
    FactTaskProductivity_HH[Visitbillable] = 0,
    FactTaskProductivity_HH[MissedVisit] = 0,
    FactTaskProductivity_HH[ServiceBillable] = 1, 
    NOT ( ISBLANK(FactTaskProductivity_HH[VisitId])
    ))

Dynamic LUPA Chart Title = 
VAR SelectedType = SELECTEDVALUE('LUPA Selector'[LUPA Type], "Actual")
RETURN
SWITCH(
    SelectedType,
    "Actual", "Patients at Actual LUPA Risk and Days to End of Period",
    "Potential", "Patients at Potential LUPA Risk and Days to End of Period",
    "Visits Needed"
)


Ttl Visits Needed (Actual vs Potential) = 
VAR SelectedType = SELECTEDVALUE('LUPA Selector'[LUPA Type], "Actual")
RETURN
SUMX(
    ADDCOLUMNS(
        --table of unique combos (bpKey)
        SUMMARIZE(
            dimBillingPeriod_HH,
            dimBillingPeriod_HH[Episode ID],
            dimBillingPeriod_HH[BillingPeriodSequence]
        ),
        "@Threshold", [LUPAThreshold],
        "@Completed", [Ttl Completed_Open],
        "@Scheduled", [Ttl Scheduled],
        "@Requested", [Ttl Requests],
        "@NeededActual", [LUPAThreshold] - [Ttl Completed_Open],
        "@NeededPotential", [LUPAThreshold] - ([Ttl Completed_Open] + [Ttl Scheduled] + [Ttl Requests] + [Ttl Missed]),
        "@Result",
            SWITCH(
                SelectedType,
                "Actual",    IF([LUPAThreshold] - [Ttl Completed_Open] > 0, [LUPAThreshold] - [Ttl Completed_Open], BLANK()),
                "Potential", IF([LUPAThreshold] - ([Ttl Completed_Open] + [Ttl Scheduled] + [Ttl Requests] ) > 0,
                                [LUPAThreshold] - ([Ttl Completed_Open] + [Ttl Scheduled] + [Ttl Requests] ),
                                BLANK()),
                BLANK()
            )
    ),--switch logic is applied
    [@Result]
)


Visits in Episodes Incl Requests = 
VAR TotalRequested =
    SUM( Requests[Requested] )
RETURN
CALCULATE (
    SUM ( FactTaskProductivity_HH[InHomeVisits_VisitNumber] ),
    FactTaskProductivity_HH[OpenClosedFlag] = "Open",
    FactTaskProductivity_HH[VisitBetweenSOEandEOE] = 1,
    FILTER (
        FactTaskProductivity_HH,
        NOT (
            FactTaskProductivity_HH[Visit Status]
                IN {
                    "OFFICE REASSIGNED",
                    "RESCHEDULED",
                    "MISSED",
                    "NON-ADMIT",
                    "DECLINED"
                }
        )
    ),
    USERELATIONSHIP (
        FactTaskProductivity_HH[skepisodeStartDateKey],
        DimDates[DateKey]
    )
)
    + TotalRequested


Visits in Episodes_Open Completed + Request = 
VAR __CurrentJobCode =
    MAX ( FactTaskProductivity_HH[Service Job Code] )

VAR __VisitsCompleted =
    CALCULATE(
        SUM(FactTaskProductivity_HH[InHomeVisits_VisitNumber]),
        FactTaskProductivity_HH[Visit Status] = "COMPLETED",
        FactTaskProductivity_HH[OpenClosedFlag] = "Open",
        FactTaskProductivity_HH[VisitBetweenSOEandEOE] = 1,
        USERELATIONSHIP(
            FactTaskProductivity_HH[skepisodeStartDateKey],
            DimDates[DateKey]
        )
    )

VAR __Requested =
    CALCULATE(
        [Requested],
        Requests[JdCode] = __CurrentJobCode
    )

RETURN
    __VisitsCompleted + __Requested


VPE - AIDE_Open Exclusion = 
VAR __CurrentJobCode = "AIDE"


VAR __Visits =
    CALCULATE(
        [Visits in Episodes_Open Exclusion],
        FactTaskProductivity_HH[Service Job Code] = __CurrentJobCode
    )


VAR __Requested =
    CALCULATE(
        [Requested],
        Requests[JdCode] = __CurrentJobCode
    )


VAR __EpisodesStarting =
    CALCULATE(
        [Episodes Starting_Open],
        FactTaskProductivity_HH[Service Job Code] = __CurrentJobCode,
        USERELATIONSHIP(
            FactTaskProductivity_HH[skepisodeStartDateKey],
            DimDates[DateKey]
        )
    )

RETURN
DIVIDE(
    __Visits + __Requested,
    __EpisodesStarting
)
